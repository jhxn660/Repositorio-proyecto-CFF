DROP DATABASE corralejo_fast_food;
CREATE DATABASE corralejo_fast_food;
USE corralejo_fast_food;

CREATE TABLE roles (
  id_rol INT PRIMARY KEY,
  rol_nombre VARCHAR(50) NOT NULL
);

CREATE TABLE categorias (
  id_categoria INT PRIMARY KEY,
  categoria_nombre VARCHAR(100) NOT NULL,
  categoria_tipo VARCHAR(20) NOT NULL
);

CREATE TABLE usuarios (
  id_usuario INT AUTO_INCREMENT PRIMARY KEY,
  id_rol INT NOT NULL,
  usuario_nombre VARCHAR(100),
  usuario_correo VARCHAR(100),
  usuario_contraseña CHAR(64),
  usuario_salt VARCHAR(32),
  usuario_telefono VARCHAR(20),
  usuario_estado VARCHAR(20) DEFAULT 'ACTIVO',
  empleado_rol VARCHAR(30),
  FOREIGN KEY (id_rol) REFERENCES roles(id_rol)
);

CREATE TABLE metas_indicadores (
  id_meta INT AUTO_INCREMENT PRIMARY KEY,
  id_usuario INT NOT NULL, 
  meta_nombre VARCHAR(100) NOT NULL,
  meta_periodo VARCHAR(20) NOT NULL,
  meta_valor_esperado DECIMAL(12,2) NOT NULL,
  FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE productos (
  id_producto INT PRIMARY KEY,
  id_categoria INT NOT NULL,
  id_usuario INT NOT NULL, 
  producto_nombre VARCHAR(100) NOT NULL,
  producto_descripcion VARCHAR(255),
  producto_precio DECIMAL(10,2) NOT NULL,
  FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria),
  FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE insumos (
  id_insumo INT PRIMARY KEY,
  id_categoria INT NOT NULL,
  id_usuario INT NOT NULL, 
  insumo_nombre VARCHAR(100) NOT NULL,
  insumo_stock DECIMAL(10,2) NOT NULL,
  insumo_medida VARCHAR(50) NOT NULL,
  insumo_costo DECIMAL(10,2) NOT NULL,
  FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria),
  FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE producto_insumo (
  id_producto_insumo INT PRIMARY KEY,
  id_producto INT NOT NULL,
  id_insumo INT NOT NULL,
  cantidad_requerida DECIMAL(10,2) NOT NULL,
  FOREIGN KEY (id_producto) REFERENCES productos(id_producto),
  FOREIGN KEY (id_insumo) REFERENCES insumos(id_insumo)
);

CREATE TABLE pedidos (
  id_pedido INT PRIMARY KEY,
  id_usuario INT NOT NULL, 
  pedido_fecha DATETIME NOT NULL,
  pedido_estado VARCHAR(50) NOT NULL DEFAULT 'PENDIENTE',
  pedido_direccion VARCHAR(255),
  pedido_total DECIMAL(10,2) NOT NULL,
  FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE pedido_detalle (
  id_pedido_detalle INT PRIMARY KEY,
  id_pedido INT NOT NULL,
  id_producto INT NOT NULL,
  cantidad INT NOT NULL,
  precio_unitario DECIMAL(10,2) NOT NULL,
  sumtotal DECIMAL(12, 2) NOT NULL DEFAULT 0.00,
  FOREIGN KEY (id_pedido) REFERENCES pedidos(id_pedido),
  FOREIGN KEY (id_producto) REFERENCES productos(id_producto)
);

CREATE TABLE ventas (
  id_venta INT PRIMARY KEY,
  venta_fecha DATETIME NOT NULL,
  venta_total DECIMAL(10,2) NOT NULL,
  ticket_qr VARCHAR(255)
);

CREATE TABLE venta_detalle (
  id_venta_detalle INT PRIMARY KEY,
  id_pedido INT NOT NULL,
  id_venta INT NOT NULL,
  FOREIGN KEY (id_pedido) REFERENCES pedidos(id_pedido),
  FOREIGN KEY (id_venta) REFERENCES ventas(id_venta)
);

CREATE TABLE liquidacion_insumo (
  id_insumo_liquidacion INT PRIMARY KEY,
  id_insumo INT NOT NULL,
  id_venta INT NOT NULL,
  FOREIGN KEY (id_insumo) REFERENCES insumos(id_insumo),
  FOREIGN KEY (id_venta) REFERENCES ventas(id_venta)
);

CREATE TABLE comentarios (
  id_comentario INT PRIMARY KEY,
  id_usuario INT NOT NULL,
  ticket_qr VARCHAR(255),
  comentario_fecha DATETIME NOT NULL,
  comentario_descripcion TEXT,
  comentario_valoracion INT NOT NULL,
  comentario_respuesta TEXT,
  FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE pqrs (
  id_pqrs INT PRIMARY KEY,
  id_usuario INT NOT NULL,
  pqrs_fecha DATETIME NOT NULL,
  pqrs_tipo VARCHAR(50) NOT NULL,
  pqrs_descripcion TEXT,
  pqrs_estado VARCHAR(50) NOT NULL DEFAULT 'EN PROCESO',
  pqrs_respuesta TEXT,
  FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

CREATE TABLE auditoria (
    id_auditoria INT AUTO_INCREMENT PRIMARY KEY,
    tabla_afectada VARCHAR(50) NOT NULL,
    accion VARCHAR(255) NOT NULL,
    id_usuario INT NOT NULL,
    fecha_auditoria DATETIME DEFAULT CURRENT_TIMESTAMP
);
