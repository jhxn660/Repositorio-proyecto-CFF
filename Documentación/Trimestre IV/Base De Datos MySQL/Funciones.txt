CREATE FUNCTION `CalcularPorcentajeMeta`(p_valor_actual DECIMAL(12,2), p_valor_esperado DECIMAL(12,2)) 
RETURNS DECIMAL(5,2) DETERMINISTIC
BEGIN
    IF p_valor_esperado = 0 THEN 
        RETURN 0.00; 
    END IF;
    RETURN (p_valor_actual / p_valor_esperado) * 100;
END //
DELIMITER ;

DELIMITER //

CREATE FUNCTION GenerarQR(p_id_venta INT) 
RETURNS VARCHAR(50) 
DETERMINISTIC
BEGIN
    -- Genera un formato estándar: HEXX-QR-YYYYMMDD-0000ID
    RETURN CONCAT(
        'HEXX-QR-', 
        DATE_FORMAT(NOW(), '%Y%m%d'), 
        '-', 
        LPAD(p_id_venta, 6, '0')
    );
END //

DELIMITER ;
